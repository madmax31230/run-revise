// Code Arduino : LEDs/Buzzer + Lecture Pédalier analogique
const int PIN_WIN_LED = 13; // LED Verte ou Gyrophare
const int PIN_BUZZER  = 8;  // Buzzer passif/actif

void setup() {
  Serial.begin(9600);
  pinMode(PIN_WIN_LED, OUTPUT);
  pinMode(PIN_BUZZER, OUTPUT);
}

void loop() {
  // 1. Réception des événements du jeu 3D
  if (Serial.available() > 0) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    if (cmd == "WIN" || cmd == "BOOST") {
      digitalWrite(PIN_WIN_LED, HIGH);
      tone(PIN_BUZZER, 1200, 150);
      delay(150);
      tone(PIN_BUZZER, 1600, 250);
    } else if (cmd == "FAIL") {
      digitalWrite(PIN_WIN_LED, LOW);
      tone(PIN_BUZZER, 220, 400);
    }
  }
  // 2. Optionnel : Si pédalier branché sur A0 (Volant), A1 (Gaz), A2 (Frein)
  // Décommente la ligne suivante :
  // Serial.print("S:"); Serial.print(analogRead(A0));
  // Serial.print(",G:"); Serial.print(analogRead(A1));
  // Serial.print(",B:"); Serial.println(analogRead(A2));
  delay(30);
}
