// Prêt. Le jeu envoie WIN, FAIL, BOOST, RACE et SIREN à l'Arduino.
void setup() {
  Serial.begin(9600);
  pinMode(13, OUTPUT); // LED Verte / Gyrophare
  pinMode(8, OUTPUT);  // Buzzer
}
void loop() {
  if (Serial.available()) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    if (cmd == "WIN" || cmd == "RACE") {
      digitalWrite(13, HIGH); tone(8, 1400, 200);
    } else if (cmd == "FAIL") {
      digitalWrite(13, LOW); tone(8, 220, 350);
    }
  }
}
