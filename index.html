🔌 Connexion Arduino (Web Serial API)

Connecte ton Arduino en USB pour allumer des LEDs sur chaque bonne réponse, déclencher un gyrophare pendant le Grand Prix IA ou lire un pédalier branché sur A0/A1. 

// Code Arduino C++ prêt à téléverser (115200 bauds)
void setup() {
  Serial.begin(115200);
  pinMode(LED_BUILTIN, OUTPUT);
}
void loop() {
  if (Serial.available()) {
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    if (cmd == "WIN" || cmd == "GP_START") digitalWrite(LED_BUILTIN, HIGH);
    if (cmd == "FAIL") digitalWrite(LED_BUILTIN, LOW);
  }
}
